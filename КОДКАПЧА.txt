Процедура ОкончаниеПеретаскивания(Элемент, ПараметрыПеретаскивания, СтандартнаяОбработка)
	СтандартнаяОбработка = Ложь;
	Если Картинка1 <> Картинка2 Тогда
		ЧастьКартинки1 = Элементы.Найти(Картинка1).Картинка;
		ЧастьКартинки2 = Элементы.Найти(Картинка2).Картинка;
		Элементы.Найти(Картинка1).Картинка = ЧастьКартинки2;
		Элементы.Найти(Картинка2).Картинка = ЧастьКартинки1;
		Если Элементы.Часть1.Картинка = БиблиотекаКартинок.ОбщаяКартинка1 И
			Элементы.Часть2.Картинка = БиблиотекаКартинок.ОбщаяКартинка2 И
			Элементы.Часть3.Картинка = БиблиотекаКартинок.ОбщаяКартинка3 И
			Элементы.Часть4.Картинка = БиблиотекаКартинок.ОбщаяКартинка4 Тогда
			КапчаПройдена = Истина;
		Иначе
			КапчаПройдена = Ложь;
		КонецЕсли;
	КонецЕсли;
КонецПроцедуры
&НаКлиенте
Процедура Войти(Команда)
	Если Не ЗначениеЗаполнено(Логин) Или Не ЗначениеЗаполнено(Пароль) Тогда
		Сообщить("Введите логин или пароль");
		Возврат;
	КонецЕсли;
		
	Если ВойтиНаСервере() Тогда
		ОткрытьФорму("ОбщаяФорма.ФормаПользователя");
		ЭтаФорма.Закрыть(); 
	Иначе
		СгенерироватьКапчу();
	КонецЕсли;
КонецПроцедуры

&НаСервере
Функция ВойтиНаСервере() 
	Попытка
		ТекПользователь = Справочники.Пользователи.НайтиПоНаименованию(Логин).ПолучитьОбъект();
	Исключение                                                                                               
		Сообщить("Не найден пользователь с указанным логином"); 
		Возврат Ложь;
	КонецПопытки;
	Если ТекПользователь.Пароль <> Пароль Тогда
		Сообщить("Неверный пароль");
		ТекПользователь.Попытки = ТекПользователь.Попытки + 1;
		ТекПользователь.Записать();
		Возврат Ложь;
	КонецЕсли;
	Если КапчаПройдена = Ложь тогда
		Сообщить("Не правильно пройдена капча");
		ТекПользователь.Попытки = ТекПользователь.Попытки + 1;
		ТекПользователь.Записать();
		Возврат Ложь;
	КонецЕсли;
	Если ТекПользователь.Попытки > 5 Тогда
		Сообщить("Превышен лимит попыток обратитесь к администратору");
		ТекПользователь.Попытки = ТекПользователь.Попытки + 1;
		ТекПользователь.Записать();
		Возврат Ложь;
	КонецЕсли; 
	ТекПользователь.Попытки = 0;
	ТекПользователь.Записать();
	Возврат Истина;
КонецФункции

Процедура СгенерироватьКапчу()
 	ГенСлуч = Новый ГенераторСлучайныхЧисел();
	МассИндексов = Новый Массив;
	МассИндексов.Добавить(1);
	МассИндексов.Добавить(2);
	МассИндексов.Добавить(3);
	МассИндексов.Добавить(4);
	Для Сч = 0 По МассИндексов.ВГраница() Цикл
		СлучИндекс = ГенСлуч.СлучайноеЧисло(0, Сч);
		ВремЗнач = МассИндексов[Сч];
		МассИндексов[Сч] = МассИндексов[СлучИндекс];
		МассИндексов[СлучИндекс] = ВремЗнач;
	КонецЦикла;
	Если МассИндексов[0] = 1 И МассИндексов[1] = 2 И МассИндексов[2] = 3 И МассИндексов[3] = 4 Тогда
		ВремЗнач = МассИндексов[0];
		МассИндексов[0] = МассИндексов[1];
		МассИндексов[1] = ВремЗнач;
	КонецЕсли;
	Элементы.Часть1.Картинка = БиблиотекаКартинок["ОбщаяКартинка" + МассИндексов[0]];
	Элементы.Часть2.Картинка = БиблиотекаКартинок["ОбщаяКартинка" + МассИндексов[1]];
	Элементы.Часть3.Картинка = БиблиотекаКартинок["ОбщаяКартинка" + МассИндексов[2]];
	Элементы.Часть4.Картинка = БиблиотекаКартинок["ОбщаяКартинка" + МассИндексов[3]];
КонецПроцедуры
