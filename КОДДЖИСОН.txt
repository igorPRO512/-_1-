Процедура ЗагрузитьJSON(СтрокаJSON)
	ЧтениеJSON = Новый ЧтениеJSON;
	ЧтениеJSON.УстановитьСтроку(СтрокаJSON);
	Массив = ПрочитатьJSON(ЧтениеJSON);
	ЧтениеJSON.Закрыть();
	Для каждого Эл Из Массив Цикл
		Наименование = Эл.name;
		Справочник = Справочники.Заказчик.НайтиПоНаименованию(Наименование,Истина); 
		Если Справочник <> Справочники.Заказчик.ПустаяСсылка() Тогда
			Спр = Справочник.ПолучитьОбъект();
		Иначе
			Спр = Справочники.Заказчик.СоздатьЭлемент();
			Спр.Наименование = Наименование;
		КонецЕсли;
		    Спр.ИНН = Эл.inn;
	        Спр.Адрес = Эл.addres;
			Спр.НомерТелефона = Эл.phone;
			Спр.Продавец = Эл.salesman;
			Спр.Покупатель = Эл.buyer;
			Спр.Записать();
	КонецЦикла;
