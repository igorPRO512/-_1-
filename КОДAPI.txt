Процедура ВыполнитьЗапрос(ИмяМетода, АдресРесурса, СтрокаТелаЗапроса)
	СерверИсточник = "localhost";
	Попытка  
		Попытка
			HTTPСоединение = Новый HTTPСоединение(СерверИсточник,80,,,, 30);
			HTTPЗапрос = Новый HTTPЗапрос(АдресРесурса);
			Результат = HTTPСоединение.ВызватьHTTPМетод(ИмяМетода, HTTPЗапрос);
		Исключение
			Сообщить("Ошибка HTTP 500 Internal Server Error");  
		КонецПопытки;
		КодСостояния = Результат.КодСостояния;    
		ТелоОтвета = Результат.ПолучитьТелоКакСтроку();
		ЧтениеJSON = Новый ЧтениеJSON;
		ЧтениеJSON.УстановитьСтроку(ТелоОтвета);
		ДанныеОтвета = ПрочитатьJSON(ЧтениеJSON, Истина);
		ЧтениеJSON.Закрыть();
		МассивЗаметок = ДанныеОтвета.Получить("value");           
		ВыбранныйОбъект = МассивЗаметок[0];
		ПолученныеДанные = ВыбранныйОбъект.Получить("title_user");    
	Исключение    
		Если КодСостояния = 200 и Не ЗначениеЗаполнено(ПолученныеДанные) Тогда
			Сообщить("Ошибка HTTP 200 OK пустые данные");
		КонецЕсли;       
		Если КодСостояния = 400 Тогда
			Сообщить("Ошибка HTTP  400 Bad Request " + Строка(КодСостояния));
		КонецЕсли;
		Сообщить(ОписаниеОшибки());
	КонецПопытки;
КонецПроцедуры
